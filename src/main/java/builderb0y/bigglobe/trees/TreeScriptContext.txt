package builderb0y.bigglobe.trees;

import net.minecraft.nbt.NbtCompound;

import builderb0y.autocodec.annotations.Wrapper;
import builderb0y.bigglobe.columns.scripted.ColumnEntryRegistry;
import builderb0y.bigglobe.columns.scripted.ColumnScript;
import builderb0y.bigglobe.columns.scripted.ScriptedColumn;
import builderb0y.bigglobe.noise.Permuter;
import builderb0y.bigglobe.scripting.environments.NbtScriptEnvironment;
import builderb0y.bigglobe.scripting.environments.RandomScriptEnvironment;
import builderb0y.bigglobe.scripting.environments.StatelessRandomScriptEnvironment;
import builderb0y.bigglobe.scripting.environments.WoodPaletteScriptEnvironment;
import builderb0y.bigglobe.scripting.wrappers.entries.WoodPaletteEntry;
import builderb0y.bigglobe.trees.TreeGenerator.NotEnoughSpaceException;
import builderb0y.scripting.bytecode.FieldInfo;
import builderb0y.scripting.bytecode.LazyVarInfo;
import builderb0y.scripting.bytecode.tree.InsnTree;
import builderb0y.scripting.environments.MathScriptEnvironment;
import builderb0y.scripting.environments.MutableScriptEnvironment;
import builderb0y.scripting.parsing.*;
import builderb0y.scripting.parsing.input.ScriptUsage;
import builderb0y.scripting.util.InfoHolder;

import static builderb0y.scripting.bytecode.InsnTrees.*;

public class TreeScriptContext {

	public static final Info INFO = new Info();
	public static class Info extends InfoHolder {

		public FieldInfo
			wood_palette,
			random,
			originX,
			originY,
			originZ,
			trunk_height,
			trunk_max_radius,
			is_stump,
			trunk_fraction,
			trunk_radius,
			has_branch,
			branch_length,
			branch_max_height,
			branch_fraction,
			branch_height,
			branch_radius,
			custom;
	}

	//provided
	public final WoodPaletteEntry wood_palette;
	public final Permuter random;
	public final double originX, originY, originZ;

	public TreeScriptContext(
		WoodPaletteEntry wood_palette,
		Permuter random,
		double originX,
		double originY,
		double originZ
	) {
		this.wood_palette = wood_palette;
		this.random       = random;
		this.originX      = originX;
		this.originY      = originY;
		this.originZ      = originZ;
	}

	//trunk init
	public double trunk_height, trunk_max_radius;
	public boolean is_stump;
	public double stump_start;

	//trunk update
	public double
		//input
		trunk_fraction,
		//output
		trunk_radius;

	//branch init
	public boolean
		//output
		has_branch;
	public double
		//output
		branch_length,
		branch_max_height,
		branch_max_radius;

	//branch update
	public double
		//input
		branch_fraction,
		//output
		branch_height,
		branch_radius;

	//custom
	public NbtCompound custom;

	public static record TreeControlScripts(
		TreeUpdateScript.TrunkInit    trunk_init,
		TreeUpdateScript.TrunkUpdate  trunk_update,
		TreeUpdateScript.BranchInit   branch_init,
		TreeUpdateScript.BranchUpdate branch_update
	) {

		public void initTrunk(ScriptedColumn column, TreeScriptContext context) throws NotEnoughSpaceException {
			context.trunk_height = Double.NaN;
			context.trunk_max_radius = Double.NaN;
			context.is_stump = false;
			context.stump_start = Double.NaN;
			this.trunk_init.update(column, context);
			boolean fail = false;
			if (Double.isNaN(context.trunk_height)) {
				fail = true;
				this.trunk_init.onError(new TreeUpdateException("trunk_init script did not write to tree.height"));
			}
			if (context.is_stump) {
				if (Double.isNaN(context.stump_start)) {
					fail = true;
					this.trunk_init.onError(new TreeUpdateException("trunk_init script did not write to tree.stump_start"));
				}
			}
			if (fail) throw NotEnoughSpaceException.INSTANCE;
		}

		public void updateTrunk(ScriptedColumn column, TreeScriptContext context, double trunkFraction) throws NotEnoughSpaceException {
			context.trunk_fraction = trunkFraction;
			context.trunk_radius = Double.NaN;
			this.trunk_update.update(column, context);
			boolean fail = false;
			if (Double.isNaN(context.trunk_radius)) {
				fail = true;
				this.trunk_update.onError(new TreeUpdateException("trunk_update script did not write to tree.trunk_radius"));
			}
			if (fail) throw NotEnoughSpaceException.INSTANCE;
		}

		public void initBranch(ScriptedColumn column, TreeScriptContext context) throws NotEnoughSpaceException {
			context.has_branch = false;
			context.branch_length = Double.NaN;
			context.branch_max_height = Double.NaN;
			context.branch_max_radius = Double.NaN;
			this.branch_init.update(column, context);
			boolean fail = false;
			if (Double.isNaN(context.branch_length)) {
				fail = true;
				this.branch_init.onError(new TreeUpdateException("branch_init script did not write to tree.branch_length"));
			}
			if (fail) throw NotEnoughSpaceException.INSTANCE;
		}

		public void updateBranch(ScriptedColumn column, TreeScriptContext context, double branchFraction) throws NotEnoughSpaceException {
			context.branch_fraction = branchFraction;
			context.branch_height = Double.NaN;
			context.branch_radius = Double.NaN;
			this.branch_update.update(column, context);
			boolean fail = false;
			if (Double.isNaN(context.branch_height)) {
				fail = true;
				this.branch_update.onError(new TreeUpdateException("branch_update script did not write to tree.branch_height"));
			}
			if (Double.isNaN(context.branch_radius)) {
				fail = true;
				this.branch_update.onError(new TreeUpdateException("branch_update script did not write to tree.branch_radius"));
			}
			if (fail) throw NotEnoughSpaceException.INSTANCE;
		}
	}

	public static class TreeUpdateException extends RuntimeException {

		public TreeUpdateException(String message) {
			super(message);
		}
	}

	public static interface TreeUpdateScript extends ColumnScript {

		public abstract void update(ScriptedColumn column, TreeScriptContext context);

		@Wrapper
		public static abstract class Holder extends ColumnScript.BaseHolder<TreeUpdateScript> implements TreeUpdateScript {

			public Holder(ScriptUsage usage) {
				super(usage);
			}

			@Override
			public Class<TreeUpdateScript> getScriptClass() {
				return TreeUpdateScript.class;
			}

			public abstract void exposeContext(MutableScriptEnvironment environment);

			public void setupEnvironment(MutableScriptEnvironment environment) {
				LazyVarInfo tree = new LazyVarInfo("context", INFO.type);
				InsnTree loadRandom = getField(load(tree), INFO.random);
				this.exposeContext(
					environment
					.addAll(MathScriptEnvironment.INSTANCE)
					.addAll(StatelessRandomScriptEnvironment.INSTANCE)
					.addType("Tree", INFO.type)
					.addVariableLoad("tree", tree)
					.addFieldGet(INFO.originX.makeFinal())
					.addFieldGet(INFO.originY.makeFinal())
					.addFieldGet(INFO.originZ.makeFinal())
					.configure(NbtScriptEnvironment.createMutable())
					.addFieldGet(INFO.custom)
					.configure(RandomScriptEnvironment.create(loadRandom))
					.configure(WoodPaletteScriptEnvironment.create(loadRandom))
					.addFieldGet(INFO.wood_palette)
				);
			}

			@Override
			public void compile(ColumnEntryRegistry registry) throws ScriptParsingException {
				this.script = (
					new TemplateScriptParser<>(TreeUpdateScript.class, this.usage)
					.configureEnvironment(this::setupEnvironment)
					.parse(new ScriptClassLoader(registry.loader))
				);
			}

			@Override
			public void update(ScriptedColumn column, TreeScriptContext context) {
				try {
					this.script.update(column, context);
				}
				catch (Throwable throwable) {
					this.onError(throwable);
				}
			}
		}

		@Wrapper
		public static class TrunkInit extends Holder {

			public TrunkInit(ScriptUsage usage) {
				super(usage);
			}

			@Override
			public void exposeContext(MutableScriptEnvironment environment) {
				environment
				.addFieldGet(INFO.trunk_height)
				.addFieldGet(INFO.trunk_max_radius)
				.addFieldGet(INFO.is_stump)
				;
			}
		}

		@Wrapper
		public static class TrunkUpdate extends Holder {

			public TrunkUpdate(ScriptUsage usage) {
				super(usage);
			}

			@Override
			public void exposeContext(MutableScriptEnvironment environment) {
				environment
				.addFieldGet(INFO.trunk_height.makeFinal())
				.addFieldGet(INFO.trunk_max_radius.makeFinal())
				.addFieldGet(INFO.trunk_fraction.makeFinal())
				.addFieldGet(INFO.trunk_radius)
				;
			}
		}

		@Wrapper
		public static class BranchInit extends Holder {

			public BranchInit(ScriptUsage usage) {
				super(usage);
			}

			@Override
			public void exposeContext(MutableScriptEnvironment environment) {
				environment
				.addFieldGet(INFO.trunk_height.makeFinal())
				.addFieldGet(INFO.trunk_max_radius.makeFinal())
				.addFieldGet(INFO.trunk_fraction.makeFinal())
				.addFieldGet(INFO.trunk_radius.makeFinal())
				.addFieldGet(INFO.has_branch)
				.addFieldGet(INFO.branch_length)
				.addFieldGet(INFO.branch_max_height)
				;
			}
		}

		@Wrapper
		public static class BranchUpdate extends Holder {

			public BranchUpdate(ScriptUsage usage) {
				super(usage);
			}

			@Override
			public void exposeContext(MutableScriptEnvironment environment) {
				environment
				.addFieldGet(INFO.trunk_height.makeFinal())
				.addFieldGet(INFO.trunk_max_radius.makeFinal())
				.addFieldGet(INFO.trunk_fraction.makeFinal())
				.addFieldGet(INFO.trunk_radius.makeFinal())
				.addFieldGet(INFO.branch_length.makeFinal())
				.addFieldGet(INFO.branch_max_height.makeFinal())
				.addFieldGet(INFO.branch_fraction.makeFinal())
				.addFieldGet(INFO.branch_height)
				.addFieldGet(INFO.branch_radius)
				;
			}
		}
	}
}

package builderb0y.bigglobe.mixins;

import java.util.ArrayList;
import java.util.List;

import me.cortex.voxy.common.world.ActiveSectionTracker;
import me.cortex.voxy.common.world.WorldSection;
import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Unique;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfoReturnable;

@Mixin(WorldSection.class)
public class WorldSection_Debugging {

	@Unique
	private List<Throwable> records;

	@Inject(method = "<init>", at = @At("RETURN"))
	private void init(int lvl, int x, int y, int z, ActiveSectionTracker tracker, CallbackInfo callback) {
		this.records = new ArrayList<>();
	}

	@Inject(method = "primeForReuse", at = @At("HEAD"))
	private void reinit(CallbackInfo callback) {
		synchronized (this.records) {
			this.records.clear();
		}
	}

	@Inject(method = "tryAcquire", at = @At("HEAD"))
	private void logTryAcquire(CallbackInfoReturnable<Boolean> callback) {
		synchronized (this.records) {
			this.records.add(new Throwable("Acquired at:"));
		}
	}

	@Inject(method = "acquire(I)I", at = @At("HEAD"))
	private void logAcquire(int count, CallbackInfoReturnable<Integer> callback) {
		synchronized (this.records) {
			this.records.add(new Throwable("Acquired at:"));
		}
	}

	@Inject(method = "release", at = @At("HEAD"))
	private void logRelease(CallbackInfoReturnable<Integer> callback) {
		synchronized (this.records) {
			this.records.add(new Throwable("Released at:"));
		}
	}
}
